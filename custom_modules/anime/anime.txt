class AnimationState : public RefCounted {
    GDCLASS(AnimationState, RefCounted);
public:
    Node* node_root = nullptr;
	Ref<Animation> animation;
	float time = 0.0;
	bool never_loop = false;
	vector<TrackCache*> track_cache;

    AnimationState() {
        //this->track_cache = Array();
     }

    void _init() { }

    void init(Animation* p_animation) {
        if (p_animation != nullptr) {
            animation = Ref<Animation>(p_animation);
            int64_t track_count = animation->get_track_count();
            track_cache.resize(track_count);
            for (int i = 0; i < track_count; i++) {
                int track_type = animation->track_get_type(i);
                TrackCache* track = memnew(TrackCache);//::_new();
                track->type = track_type;
                track_cache[i] = track;
            }
        }
     }

    AnimationState* copy() {
        AnimationState* copy = memnew(AnimationState);//::_new();
        /*
        if (this->animation != nullptr) {
            copy->init(*this->animation);
        }
        */
        copy->animation = this->animation;
        copy->node_root = this->node_root;
        copy->track_cache = this->track_cache;

        /*
        // deep copy
        int64_t track_size = this->track_cache.size();
        copy->track_cache.resize(track_size);
        for (int64_t i = 0; i < track_size; i++) {
            copy->track_cache[i] = this->track_cache[i]->copy();
        }
        */

        copy->time = this->time;
        return copy;
     }

    static void _register_methods() {
        ClassDB::bind_method(D_METHOD("init"), &AnimationState::init);

    void play_animation(AnimationState* state, float delta, bool update_cache = true) {
        Ref<Animation> animation = state->animation;
        
        state->time += delta;
        
        if (state->time > state->animation->get_length()) {
            if (animation->get_loop_mode() != Animation::LOOP_NONE && state->never_loop == false) {
                state->time = 0.0;
            } else {
                state->time = state->animation->get_length();
            }
        } else if (state->time < 0.0) { // Loop backwards
            if (animation->get_loop_mode() != Animation::LOOP_NONE && state->never_loop == false) {
                state->time = state->animation->get_length();
            } else {
                state->time = 0.0;
            }
        }
        
        if (update_cache == false) {
            return;
        }

        Node* root_node = state->node_root;

        if (root_node == nullptr) {
            NodePath root_node_path = animation_player->get_root();
            root_node = animation_player->get_node_or_null(root_node_path);
            if (root_node == nullptr) {
                UtilityFunctions::print("AnimationPlayer node_root is null!");
                return;
            } else {
                state->node_root = root_node;
            }
        }
        
        int64_t track_count = animation->get_track_count();
        for (int64_t track_idx = 0; track_idx < track_count; track_idx++) {
            if (!animation->track_is_enabled(track_idx)) {
                continue;
            }

            godot::Animation::TrackType track_type = animation->track_get_type(track_idx);

            switch(track_type) {
                case godot::Animation::TrackType::TYPE_ROTATION_3D:

                            continue;
                        }

    void _blend_animation(AnimationState* state, AnimationState* blend_state, float blend_amount, enum BlendMode blend_mode, Dictionary filter) {
        // TODO: Only blends transform at the moment
        // TODO: If a track exists in one but not another how are they combined?
        // Solution: Only blend tracks that exist in state, extra tracks in blend_state, wont be blended.
        if (Math::is_equal_approx(blend_amount, 0.0f)) {
            return;
        }
        
        Ref<Animation> animation = state->animation; 
        Ref<Animation> blend_animation = blend_state->animation; 
        int64_t track_count = animation->get_track_count();
        for (int track_idx = 0; track_idx < track_count; track_idx++) {
            Animation::TrackType track_type = animation->track_get_type(track_idx);
            Animation::TrackType blend_track_type = blend_animation->track_get_type(track_idx);
            //NodePath track_path = animation->track_get_path(track_idx);

            if (track_type != blend_track_type) {
                continue;
            }
            
            switch(track_type) {
                case Animation::TrackType::TYPE_ROTATION_3D:
                {
                    /*
                    if (state->track_cache[track_idx]->bone_idx != blend_state->track_cache[track_idx]->bone_idx) {
                        Godot::print("The track idx are not aligned for these two animations. A blending cannot occur without updating the code.");
                        continue;
                    }
                    */

                   TrackCache* track = state->track_cache[track_idx];
                   TrackCache* blend_track = blend_state->track_cache[track_idx];
                    
                    float blend = Math::clamp(abs(blend_amount), 0.0f, 1.0f);

                    if (track->skeleton != nullptr) {
                        if (!filter.is_empty()) {
                            String bone_name = track->skeleton->get_bone_name(track->bone_idx);
                            if (filter.has(bone_name)) {
                                Variant weight = filter[bone_name];
                                if (weight.get_type() == Variant::FLOAT) {
                                    blend = Math::clamp(abs((float)weight), 0.0f, 1.0f);
                                    if (Math::is_zero_approx(blend)) {
                                        continue;
                                    }
                                }
                            }
                        }
                    }
                    
                    Vector3 pos = track->pos;
                    Vector3 scale = track->scale;
                    Quaternion rot = track->rot;
                    
                    if (blend_mode == BlendMode::SUB) {
                        blend = -blend;
                        blend_mode = BlendMode::ADD;
                    }

                    switch (blend_mode) {
                        case BlendMode::ADD: 
                            {
                                // Direct addition / subtraction.
                                pos += blend_track->pos * blend;
                                Quaternion q = Quaternion();
                                q = q.slerp(blend_track->rot.normalized(), blend).normalized();
                                rot = (rot * q).normalized();
                                break;						
                            }
                        case BlendMode::BLEND: 
                            {
                                pos = pos.lerp(blend_track->pos, blend);
                                rot = rot.slerp(blend_track->rot, blend).normalized();
                                break;
                            }
                    } 
                    
                    scale = scale.lerp(blend_track->scale, abs(blend));

                    track->pos = pos;
                    track->scale = scale;
                    track->rot = rot;
                }
            }
        }
    }
